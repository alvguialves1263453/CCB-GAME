-- 1. Apaga tabela bugada e recria limpa
DROP TABLE IF EXISTS online_presence;

CREATE TABLE online_presence (
  id text PRIMARY KEY,
  nickname text NOT NULL,
  avatar text,
  status text,
  status_detail text,
  view text,
  game_type text,
  updated_at timestamp with time zone DEFAULT now()
);

-- 2. DESATIVA RLS (anon consegue ver todos, por isso tava 1 só com <você>)
ALTER TABLE online_presence DISABLE ROW LEVEL SECURITY;

-- 3. Habilita Realtime pro PC ver iPhone instantâneo
ALTER PUBLICATION supabase_realtime ADD TABLE online_presence;

-- 4. Índice
CREATE INDEX idx_online_presence_updated ON online_presence(updated_at);

-- 5. Limpa fantasmas
DELETE FROM online_presence;
